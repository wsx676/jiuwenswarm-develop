```python
def _write_industry(self, data: dict, request,
                    revision_feedback=None) -> ReportDraft:
    """行业研报：板块级聚合，覆盖核心观点/竞争格局/估值/风险"""
    draft = ReportDraft()
    name = getattr(request, "name", "") or getattr(request, "target", "")
    target = getattr(request, "target", "")
    industry = data.get("industry_analysis")
    macro = data.get("macro_analysis")

    # 1. 大纲（LLM 优先，模板降级）
    outline = self._build_outline_industry(name, industry, macro)
    draft.outline = outline

    # 2. 逐段撰写
    revision = self._revision_instructions(
        (revision_feedback or {}).get("issues", []) or [])
    sections: List[Tuple[str, str]] = []
    for part in outline:
        title, desc = part["part_title"], part["part_desc"]
        material = self._materials_for_industry(
            title, data=data, name=name, target=target,
            industry=industry, macro=macro)
        body = self._write_section(
            name, title, desc, material,
            context=self._digest(sections), revision=revision)
        sections.append((title, body))

    # 3. 装配（含行业对比图表 + 来源清单 + 免责声明）
    charts = self._build_industry_charts(data, target)
    draft.charts = charts
    draft.content = self._assemble_industry(
        sections, charts, data, name, target, request)
    draft.citations = self._collect_sources_industry(data, target)
    draft.claims = self._build_claims_industry(data, industry, macro)
    return draft
```

```python
def _materials_for_industry(self, title, *, data, name, target,
                            industry, macro):
    """行业研报每段的 JSON 材料（程序化拼接 + 段末「数据来源」）"""
    industry_dict = industry.to_dict() if industry else {}
    macro_dict = macro.to_dict() if macro else {}
    if "核心观点" in title:
        payload = {
            "板块": name,
            "景气度": industry_dict.get("prosperity", {}).get("level", ""),
            "景气分": industry_dict.get("prosperity", {}).get("score", ""),
            "板块洞察": industry_dict.get("insights", [])[:5],
        }
    elif "投资结论" in title:
        # 板块级评级规则
        level = industry_dict.get("prosperity", {}).get("level", "")
        rating = "超配" if level == "景气向上" else \
                 "低配" if level == "景气承压" else "标配"
        payload = {"板块评级": rating,
                   "决策依据": f"板块景气度={level}",
                   "板块洞察": industry_dict.get("insights", [])[:3]}
    elif "竞争格局" in title:
        payload = {"竞争排名": industry_dict.get("competition", {})}
    elif "估值" in title:
        payload = {"板块平均PE": industry_dict.get("avg_pe"),
                   "估值分位": industry_dict.get("pe_quantile")}
    elif "宏观" in title:
        payload = {"宏观指标": macro_dict.get("indicators", {}),
                   "板块传导": macro_dict.get("sector_impact", {})}
    elif "风险" in title:
        payload = {"板块风险信号": industry_dict.get("risk_signals", [])}
    else:
        payload = {"说明": "本段由程序生成"}
    return json.dumps(payload, ensure_ascii=False, default=str)[:2600]
```

```python
def _assemble_industry(self, sections, charts, data, name, target, request):
    """装配行业研报：标题块 + 章节正文 + 板块对比图 + 来源 + 免责声明"""
    period = getattr(request, "period", "") or "最新"
    lines = [f"# {name}行业研究分析报告", ""]
    lines += ["| 属性 | 值 |", "|------|-----|",
              f"| 报告类型 | 行业研报 |",
              f"| 报告日期 | {period} |", ""]
    by_title = {t: b for t, b in sections}
    order = [t for t, _ in sections]

    for title in order:
        body = by_title[title]
        if "数据来源" in title:
            continue
        lines.append(f"## {title}")
        lines.append("")
        lines.append(body)
        # 板块对比图随"竞争格局"章节
        if "竞争格局" in title:
            for c in charts:
                if c.chart_type == "bar":
                    lines += chart_md(c)
        lines.append("")

    # 来源清单
    sources = self._collect_sources_industry(data, target)
    lines.append("## 八、数据来源")
    lines.append("")
    lines += [f"- {s}" for s in sources]
    lines.append("数据来源：以上权威渠道来源清单汇总")
    lines += ["", "---",
              "*免责声明：本报告由 AI Agent 自动生成，仅供参考，"
              "不构成投资建议。*"]
    content = "\n".join(lines)
    content, issues = self._localize_images(content)
    data["image_issues"] = issues
    return content
```

```python
def _build_industry_charts(self, data, sector):
    """板块内 top 5 净利润横向对比柱图（无数据时返回空）"""
    industry = data.get("industry_analysis")
    if not industry:
        return []
    competition = industry.competition or {}
    peers = competition.get("peers", []) or []
    if not peers:
        return []
    top = sorted(peers, key=lambda p: p.get("net_profit", 0) or 0,
                 reverse=True)[:5]
    try:
        from generators.chart_generator import ChartGenerator
        gen = ChartGenerator()
        # 构造营收/净利润双柱图
        return [gen.generate_sector_bar(
            sector_name=sector, peers=top)]
    except Exception as e:
        logger.warning("板块对比图生成失败: %s", e)
        return []
```

```python
def _collect_sources_industry(self, data, sector):
    sources = []
    pool = data.get("pool") or {}
    sources.append(f"行业分组：组委会公司池（{sector} 板块）")
    macro = data.get("macro_analysis")
    if macro and macro.indicators:
        sources.append("宏观数据：国家统计局（经 akshare 接口）")
    industry = data.get("industry_analysis")
    if industry and industry.news_items:
        seen = set()
        for it in industry.news_items[:10]:
            src = it.get("source", "")
            if src and src not in seen:
                seen.add(src)
                sources.append(f"新闻资讯：{src}")
    if data.get("knowledge_chunks"):
        sources.append(
            "行业研究方法论：财务方法论知识库（自沉淀方法论文档）")
    return sources
```

## 四、应急方案

### 4.1 LLM 不可用时的最快路径

如果 `LLMClient` 不可用（无 API_KEY 等），**不要等待**——直接用规则模板降级产出 1 份结构完整的报告：

```bash
# LLM 不可用时直接跑（_template_section 全程序化拼接）
python run_report.py industry --name 消费板块 --save
```

`_template_section` 在 `SECTION_SOURCE_INDUSTRY` 字典保证每段末带「数据来源」，引用率 100% 合规。

### 4.2 industry_analyzer 数据空

若 `_peer_metrics` 返回空（板块内标的财报未采集），行业研报仍能产出但内容偏空：
- 立即跑 `python run_report.py research --stage collect --save`
- 等数据落盘后再跑 industry

### 4.3 时间超时的最小路径

若只剩 1 小时，按以下优先级跑：

1. **必须**：`industry --name 消费板块 --save`（板块已采集过）
2. **必须**：`macro --period 2026Q2 --save`
3. 可选：再多跑 1-2 个板块的 industry 报告
